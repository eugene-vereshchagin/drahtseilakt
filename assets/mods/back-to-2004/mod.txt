# Back to 2004: undoes our rendering fixes, so the game looks as it did on a PC of the year of its release: a low
# resolution scaled up with hard pixels in a 4:3 frame, textures without mipmaps or anisotropy (the distant shimmer),
# a frame cap and, if you dare, 16-bit colour. Only the look changes; the run stays classic, replays play the same.

on frame
if opt.resolution == 0
do render.scale 640 480

on frame
if opt.resolution == 1
do render.scale 800 600

on frame
if opt.resolution == 2
do render.scale 1024 768

on frame
if opt.bars
do render.aspect 4:3

on frame
if opt.filter == 1
do render.filter bilinear

on frame
if opt.filter == 2
do render.filter nearest

on frame
if opt.cap == 1
do render.fps_cap 60

on frame
if opt.cap == 2
do render.fps_cap 30

on frame
if opt.banding
do render.colour_bits 16
